def criar_conexao(nome_banco):
    """Cria conexão com o banco e ativa suporte a chaves estrangeiras."""
    try:
        conn = sqlite3.connect(nome_banco)
        conn.execute("PRAGMA foreign_keys = ON")  # Ativa FK
        print(f"Conectado ao banco: {nome_banco}")
        return conn
    except sqlite3.Error as e:
        print(f"Erro ao conectar: {e}")
        return None

def criar_tabelas(conn):
    """Cria tabelas com chave estrangeira."""
    try:
        cursor = conn.cursor()

        # Tabela de usuários
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS usuarios (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nome TEXT NOT NULL
            )
        """)

        # Tabela de pedidos com chave estrangeira para usuarios.id
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS pedidos (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                descricao TEXT NOT NULL,
                usuario_id INTEGER NOT NULL,
                FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
                    ON DELETE CASCADE
                    ON UPDATE CASCADE
            )
        """)

        conn.commit()
        print("Tabelas criadas com sucesso.")
    except sqlite3.Error as e:
        print(f"Erro ao criar tabelas: {e}")

def inserir_usuario(conn, nome):
    """Insere um usuário e retorna seu ID."""
    try:
        cursor = conn.cursor()
        cursor.execute("INSERT INTO usuarios (nome) VALUES (?)", (nome,))
        conn.commit()
        return cursor.lastrowid
    except sqlite3.Error as e:
        print(f"Erro ao inserir usuário: {e}")

def inserir_pedido(conn, descricao, usuario_id):
    """Insere um pedido vinculado a um usuário."""
    try:
        cursor = conn.cursor()
        cursor.execute("INSERT INTO pedidos (descricao, usuario_id) VALUES (?, ?)", (descricao, usuario_id))
        conn.commit()
        print("Pedido inserido com sucesso.")
    except sqlite3.IntegrityError:
        print("Erro: usuário não existe (violação de chave estrangeira).")
    except sqlite3.Error as e:
        print(f"Erro ao inserir pedido: {e}")

def listar_pedidos(conn):
    """Lista pedidos com nome do usuário."""
    try:
        cursor = conn.cursor()
        cursor.execute("""
            SELECT pedidos.id, pedidos.descricao, usuarios.nome
            FROM pedidos
            JOIN usuarios ON pedidos.usuario_id = usuarios.id
        """)
        for row in cursor.fetchall():
            print(row)
    except sqlite3.Error as e:
        print(f"Erro ao listar pedidos: {e}")

def main():
    conn = criar_conexao("loja.db")
    if conn:
        criar_tabelas(conn)

        # Inserindo usuários
        id_alice = inserir_usuario(conn, "Alice")
        id_bob = inserir_usuario(conn, "Bob")

        # Inserindo pedidos
        inserir_pedido(conn, "Notebook", id_alice)
        inserir_pedido(conn, "Smartphone", id_bob)

        # Tentativa de inserir pedido para usuário inexistente
        inserir_pedido(conn, "Tablet", 999)

        print("\nPedidos cadastrados:")
        listar_pedidos(conn)

        conn.close()

if __name__ == "__main__":
    main()

import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders

# Define the sender and receiver email addresses
fromaddr = "sender@example.com"
toaddr = "receiver@example.com"

# Create an instance of MIMEMultipart
msg = MIMEMultipart()

# Set the sender, receiver, and subject of the email
msg['From'] = fromaddr
msg['To'] = toaddr
msg['Subject'] = "Subject of the Mail"

# Create the body of the email
body = "This is the body of the email."

# Attach the body to the email
msg.attach(MIMEText(body, 'plain'))

# Open the file to be sent
filename = "example.txt"
attachment = open("path/to/example.txt", "rb")

# Create a MIMEBase instance
part = MIMEBase('application', 'octet-stream')

# Set the payload of the attachment
part.set_payload(attachment.read())

# Encode the attachment in base64
encoders.encode_base64(part)

# Add header to the attachment
part.add_header('Content-Disposition', f"attachment; filename= {filename}")

# Attach the instance to the email
msg.attach(part)

# Create an SMTP session
server = smtplib.SMTP('smtp.gmail.com', 587)
server.starttls()

# Login to the email account
server.login(fromaddr, "password")

# Convert the MIMEMultipart object to a string
text = msg.as_string()

# Send the email
server.sendmail(fromaddr, toaddr, text)

# Terminate the SMTP session
server.quit()
